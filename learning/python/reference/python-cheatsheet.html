Python Cheatsheet

Quick reference — syntax, built-ins, and common patterns.

Variables & Types

Basic types
x = 5            # int
y = 3.14         # float
name = "Ada"     # str
ok = True        # bool
nothing = None   # NoneType
Type checking
type(x)              # <class 'int'>
isinstance(x, int)   # True

Strings

f-strings
name = "Ada"
f"Hello, {name}!"
f"{3.14159:.2f}"     # '3.14'
Common methods
"a,b,c".split(",")     # ['a','b','c']
",".join(["a","b"])    # 'a,b'
"  hi ".strip()        # 'hi'
"hi".replace("h","H")  # 'Hi'
"hi".upper() / .lower()
Slicing
s = "hello"
s[1:4]     # 'ell'
s[::-1]    # 'olleh' (reverse)

Data Structures

List
lst = [1, 2, 3]
lst.append(4)
lst.pop()          # removes & returns last
lst.insert(0, 0)
lst[0], lst[-1]    # first, last
Dict
d = {"a": 1, "b": 2}
d["a"]             # 1
d.get("c", 0)      # 0 (default)
d.keys() / .values() / .items()
Set
s = {1, 2, 3}
s.add(4)
s1 & s2   # intersection
s1 | s2   # union
Tuple
t = (1, 2, 3)   # immutable
a, b, c = t     # unpacking
Comprehensions
[x*2 for x in range(5)]
{x: x*2 for x in range(3)}
[x for x in range(10) if x % 2 == 0]

Control Flow

Conditionals
if x > 0:
    print("pos")
elif x == 0:
    print("zero")
else:
    print("neg")
For loop
for i in range(5):
    print(i)

for i, val in enumerate(lst):
    print(i, val)
While loop
while x > 0:
    x -= 1
Break / continue
for i in range(10):
    if i == 5: break
    if i % 2 == 0: continue

Functions

Definition
def greet(name, greeting="Hi"):
    return f"{greeting}, {name}!"

greet("Ada")               # 'Hi, Ada!'
greet("Ada", "Hello")
Lambda
square = lambda x: x * x
sorted(lst, key=lambda x: -x)
*args / **kwargs
def f(*args, **kwargs):
    print(args, kwargs)

f(1, 2, a=3)   # (1, 2) {'a': 3}

Error Handling

Try / except
try:
    1 / 0
except ZeroDivisionError as e:
    print(e)
finally:
    print("done")
Raising errors
raise ValueError("bad input")

Classes (OOP)

Basic class
class Dog:
    def __init__(self, name):
        self.name = name

    def bark(self):
        return f"{self.name} says woof"

d = Dog("Rex")
d.bark()
Inheritance
class Puppy(Dog):
    def __init__(self, name):
        super().__init__(name)

File I/O

Reading/writing
with open("f.txt") as f:
    content = f.read()

with open("f.txt", "w") as f:
    f.write("hello")

Useful Built-ins

Common functions
len(lst), sum(lst)
max(lst), min(lst)
sorted(lst, reverse=True)
zip(list1, list2)
map(func, lst), filter(func, lst)
